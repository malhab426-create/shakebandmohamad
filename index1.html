-- 1. إنشاء جدول السيارات (مع حقول الملكية الجديدة) CREATE TABLE IF NOT EXISTS public.vehicles ( id TEXT PRIMARY KEY, plate TEXT NOT NULL, make TEXT NOT NULL, model TEXT NOT NULL, year INTEGER, type TEXT, status TEXT DEFAULT 'active', odometer INTEGER DEFAULT 0, tank INTEGER DEFAULT 70, fuel_level INTEGER DEFAULT 50, driver TEXT, next_service_km INTEGER DEFAULT 0, tc_no TEXT, owner TEXT, nationality TEXT, registration_date DATE, registration_expiry DATE, insurance_expiry DATE, policy_no TEXT, passengers INTEGER, origin TEXT, color TEXT, gvw INTEGER, empty_weight INTEGER, engine_no TEXT, chassis_no TEXT, created_at DATE DEFAULT CURRENT_DATE ); -- 2. إنشاء جدول السائقين CREATE TABLE IF NOT EXISTS public.drivers ( id TEXT PRIMARY KEY, name TEXT NOT NULL, phone TEXT NOT NULL, license_no TEXT, license_expiry DATE, vehicle_id TEXT REFERENCES public.vehicles(id) ON DELETE SET NULL, status TEXT DEFAULT 'active' ); -- 3. إنشاء جدول الصيانة CREATE TABLE IF NOT EXISTS public.maintenance ( id TEXT PRIMARY KEY, request_no TEXT, vehicle_id TEXT REFERENCES public.vehicles(id) ON DELETE SET NULL, garage_name TEXT, garage_address TEXT, garage_contact_name TEXT, garage_contact_phone TEXT, maintenance_date DATE, delivery_date DATE, delivery_time TEXT, receipt_date DATE, receipt_time TEXT, employee_name TEXT, employee_phone TEXT, maintenance_type TEXT DEFAULT 'periodic', items JSONB DEFAULT '[]'::jsonb, final_items JSONB DEFAULT '[]'::jsonb, notes TEXT, close_notes TEXT, acknowledged BOOLEAN DEFAULT FALSE, attachments JSONB DEFAULT '{"damaged":[],"quote":[],"replaced":[],"invoice":[]}'::jsonb, odometer INTEGER DEFAULT 0, cost NUMERIC DEFAULT 0, status TEXT DEFAULT 'open', stage TEXT DEFAULT 'open', closed_at DATE, request_id TEXT ); -- 4. إنشاء جدول التعقب CREATE TABLE IF NOT EXISTS public.tracking ( id TEXT PRIMARY KEY, vehicle_id TEXT REFERENCES public.vehicles(id) ON DELETE CASCADE, location TEXT, lat NUMERIC, lng NUMERIC, speed INTEGER DEFAULT 0, status TEXT DEFAULT 'idle', zone TEXT, parked_since TIMESTAMPTZ, at TIMESTAMPTZ DEFAULT NOW() ); -- 5. إنشاء سجل التعقب CREATE TABLE IF NOT EXISTS public."trackingLog" ( id TEXT PRIMARY KEY, vehicle_id TEXT, location TEXT, speed INTEGER, status TEXT, at TIMESTAMPTZ DEFAULT NOW() ); -- 6. إنشاء جدول المهمات CREATE TABLE IF NOT EXISTS public.tasks ( id TEXT PRIMARY KEY, title TEXT, vehicle_id TEXT REFERENCES public.vehicles(id) ON DELETE SET NULL, driver TEXT, "from" TEXT, "to" TEXT, start_date DATE, end_date DATE, priority TEXT DEFAULT 'medium', status TEXT DEFAULT 'scheduled' ); -- 7. إنشاء جدول الطلبات CREATE TABLE IF NOT EXISTS public.requests ( id TEXT PRIMARY KEY, vehicle_id TEXT REFERENCES public.vehicles(id) ON DELETE SET NULL, requested_by TEXT, date DATE, priority TEXT DEFAULT 'medium', description TEXT, status TEXT DEFAULT 'new' ); -- 8. إنشاء جدول الوثائق CREATE TABLE IF NOT EXISTS public.documents ( id TEXT PRIMARY KEY, vehicle_id TEXT REFERENCES public.vehicles(id) ON DELETE CASCADE, type TEXT, name TEXT, number TEXT, issue_date DATE, expiry_date DATE, issuer TEXT, notes TEXT ); -- 9. إنشاء جدول الوقود CREATE TABLE IF NOT EXISTS public.fuel ( id TEXT PRIMARY KEY, vehicle_id TEXT REFERENCES public.vehicles(id) ON DELETE CASCADE, date DATE, liters NUMERIC, price_per_liter NUMERIC, cost NUMERIC, odometer INTEGER, station TEXT, receipt_image TEXT DEFAULT '' ); -- 10. إنشاء جدول المخالفات CREATE TABLE IF NOT EXISTS public.violations ( id TEXT PRIMARY KEY, vehicle_id TEXT REFERENCES public.vehicles(id) ON DELETE CASCADE, driver TEXT, type TEXT DEFAULT 'salik', date DATE, amount NUMERIC DEFAULT 0, description TEXT, location TEXT, reference_no TEXT, status TEXT DEFAULT 'unpaid' ); -- 11. إنشاء جدول المستخدمين (Profiles) CREATE TABLE IF NOT EXISTS public.profiles ( id UUID PRIMARY KEY REFERENCES auth.users ON DELETE CASCADE, email TEXT, full_name TEXT, role TEXT DEFAULT 'user' CHECK (role IN ('admin','user')), created_at TIMESTAMPTZ DEFAULT NOW() );
